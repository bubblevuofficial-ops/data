Q3) write only convert() and int main ... to convert char 'Z' to
int 90.
ANSWER:
#include<iostream>
using namespace std;

template<typename T>
class Convertor
{	
	private:
		T first;
	public:
		Convertor(T first) {
			this -> first = first;
		}
		
		void convert() {
			cout << int(first);
		}
};

int main() {
	
	Convertor<char> obj('Z');
	obj.convert();
	
	return 0;
}
